# Function to get unique ICD codes from a single string
get_unique_ICD_from_string <- function(ICD_string) {
  # Split string by comma or space
  ICD_vector <- unlist(strsplit(ICD_string, split = "[, \n]+"))
  
  # Trim whitespace from each element
  ICD_vector <- trimws(ICD_vector)
  
  # Convert to character (in case of numeric-like codes)
  ICD_vector <- as.character(ICD_vector)
  
  # Return unique values
  unique(ICD_vector)
}

# Function to expand wildcard codes
expand_wildcards <- function(wildcard_string, master_codes) {
  wildcard_vector <- unlist(strsplit(wildcard_string, split = "[, \n]+"))
  wildcard_vector <- trimws(wildcard_vector)
  
  expanded <- map(wildcard_vector, function(pat) {
    regex_pat <- paste0("^", str_replace_all(pat, "x", ""))
    master_codes$code[str_detect(master_codes$code, regex_pat)]
  }) %>% unlist()
  
  unique(expanded)
}

wildcard_expanded <- expand_wildcards(wildcards, icd10codes)

# make into tibble for joining
combined_codes_df <- tibble(code = combined_codes_unique)

# join with icd10codes on the code
lookup <- combined_codes_df %>%
  left_join(icd10codes, by = c("code")) %>%
  rename(`Number / Range` = code,
         `Full Description` = description) %>%
  mutate(
    `Procedure type` = case_when(
      str_detect(`Full Description`, regex("Bypass Coronary Artery", ignore_case = TRUE)) ~ "Coronary Artery Bypass Graft",
      str_detect(`Full Description`, regex("Mitral Valve", ignore_case = TRUE)) ~ "Mitral Valve repair/replacement",
      str_detect(`Full Description`, regex("Aortic Valve", ignore_case = TRUE)) ~ "Aortic Valve repair/replacement",
      str_detect(`Full Description`, regex("Tricuspid Valve", ignore_case = TRUE)) ~ "Tricuspid Valve repair/replacement",
      str_detect(`Full Description`, regex("Pulmonary Valve", ignore_case = TRUE)) ~ "Pulmonary Valve repair/replacement",
      TRUE ~ str_extract(`Full Description`, "^[A-Za-z]+")
    ),
    Approach = str_extract(`Full Description`,
                           "(Open Approach|Percutaneous Endoscopic Approach|Percutaneous Approach)"),
        Replacement = str_detect(`Full Description`, regex("Replacement", ignore_case = TRUE))

  )

write.csv(lookup, "output/ICD10_cardiacSGY_lookup.csv", row.names = FALSE)
