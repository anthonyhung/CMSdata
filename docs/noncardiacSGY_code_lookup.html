# Function to extract ICD-10-PCS codes
extract_icd10pcs <- function(text) {
  # ICD-10-PCS: 7 chars, A-Z (except I,O) and 0-9
  pattern <- "\\b[0-9A-HJ-NP-Z]{7}\\b"
  
  codes <- str_extract_all(text, pattern)[[1]]
  unique(codes)
}

# Function to expand wildcard codes
expand_wildcards <- function(wildcard_string, master_codes) {
  wildcard_vector <- unlist(strsplit(wildcard_string, split = "[, \n]+"))
  wildcard_vector <- trimws(wildcard_vector)
  
  expanded <- map(wildcard_vector, function(pat) {
    regex_pat <- paste0("^", str_replace_all(pat, "x", ""))
    master_codes$code[str_detect(master_codes$code, regex_pat)]
  }) %>% unlist()
  
  unique(expanded)
}
