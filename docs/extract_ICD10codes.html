Define and implement functions 1. extract_icd10pcs extracts PCS codes from free text, leveraging the structure of the codes as 7 length alphanumeric strings that do not use letters {I,O} 2. classify_cardic_codes sorts the extracted PCS codes into: CABG (starts with 021) vs individual valve (based on the 4th position character in the code) and then from there into open vs percutaneous approach based on the 5th position character

# Function to extract ICD-10-PCS codes
extract_icd10pcs <- function(text) {
  # ICD-10-PCS: 7 chars, A-Z (except I,O) and 0-9
  pattern <- "\\b[0-9A-HJ-NP-Z]{7}\\b"
  
  codes <- str_extract_all(text, pattern)[[1]]
  unique(codes)
}

# Function to place the ICD-10-PCS codes into the correct body part/approach category
classify_cardiac_codes <- function(codes) {
  
  df <- data.frame(
    code = codes,
    body_part = substr(codes, 4, 4),
    approach = substr(codes, 5, 5),
    stringsAsFactors = FALSE
  )
  
  # Identify CABG separately: they all start with 021
  df$category <- ifelse(substr(df$code, 1, 3) == "021",
                        # CABG classification
                        ifelse(df$approach == "0", "CABGOPEN",
                               ifelse(df$approach %in% c("3", "4"), "CABGPERC", "CABGOther")),
                        NA)
  
  # Valves: map by body part character position
  valve_map <- c("F" = "Aortic", "G" = "Mitral", "H" = "Pulmonic", "J" = "Tricuspid")
  df$valve <- valve_map[df$body_part]
  
  df$type <- case_when(
    df$approach == "0" ~ "OPEN",
    df$approach %in% c("3", "4") ~ "PERC",
    TRUE ~ NA
  )
  
  df$category <- ifelse(is.na(df$category) & !is.na(df$valve),
                        paste0(df$valve, df$type),
                        df$category)
  
  # Collapse to summary dataframe
  summary_df <- df %>%
    filter(!is.na(category)) %>%
    group_by(category) %>%
    summarise(codes = paste(unique(code), collapse = ", "),
              .groups = "drop")
  
  return(summary_df)
}
